Запрос: число инцидентов в самом плохом регионе по SLA по месяцам
SQL: SELECT
    date_trunc('month', sr.created_at) AS month,
    COUNT(*) AS incidents
FROM service_requests sr
JOIN service_objects so
    ON sr.object_id = so.id
JOIN clients c
    ON so.client_id = c.id
JOIN regions r
    ON c.region_id = r.id
WHERE sr.request_type = 'Инцидент'
  AND sr.status IN ('Открыта', 'В работе')
  AND r.name = (
        SELECT region
        FROM mart_sla_region
        GROUP BY region
        ORDER BY 1 - AVG(sla_rate) DESC
        LIMIT 1
  )
GROUP BY month
ORDER BY month;

Запрос: среднее время ремонта банкоматов по 5 самым прибыльным клиентам, какие имена клиентов, их телефон и контактное лицо
SQL: SELECT 
    c.name AS client_name, 
    cc.contact_name, 
    cc.phone, 
    AVG(sr.duration_hours) AS average_repair_time,
    f.revenue,
    f.costs,
    (f.revenue - f.costs) AS profit
FROM 
    clients c
JOIN 
    service_objects so ON c.id = so.client_id
JOIN 
    service_requests sr ON so.id = sr.object_id
JOIN 
    client_contacts cc ON c.id = cc.client_id
LEFT JOIN 
    finance f ON c.id = f.client_id AND f.month = (
        SELECT MAX(month) FROM finance WHERE client_id = c.id
    )
WHERE 
    so.type = 'Банкомат' 
    AND sr.request_type = 'Инцидент' 
    AND sr.status = 'Закрыта'
GROUP BY 
    c.name, cc.contact_name, cc.phone, f.revenue, f.costs
ORDER BY 
    (f.revenue - f.costs) DESC
LIMIT 5;

Запрос: какие инженеры больше всего работали с кассами в прошлом месяце, имя региона, квалификация и телефон каждого инженера
SQL: SELECT 
  e.full_name,
  r.name AS region_name,
  e.skill_level,
  -- добавляем первый телефон инженера
  (SELECT phone FROM engineer_contacts ec WHERE ec.engineer_id = e.id LIMIT 1) AS phone
FROM engineers e
JOIN engineer_skills es ON e.id = es.engineer_id
JOIN regions r ON e.region_id = r.id
JOIN service_requests sr ON e.id = sr.engineer_id
JOIN service_objects so ON sr.object_id = so.id
WHERE 
  es.object_type = 'Касса'
  AND sr.created_at >= date_trunc('month', current_date - interval '1 month')
  AND sr.created_at < date_trunc('month', current_date)
GROUP BY e.id, e.full_name, r.name, e.skill_level
ORDER BY COUNT(*) DESC
LIMIT 200;
